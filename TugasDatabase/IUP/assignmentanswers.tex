\section{Entity Relationship Diagram}

The database uses six main tables: \texttt{students}, \texttt{professors}, \texttt{courses}, \texttt{prerequisites}, \texttt{AvailableClasses}, and \texttt{CourseGrades}. Students, professors, and courses are identified by their NIM, professor ID, and course ID.

\texttt{AvailableClasses} records which class a student joins, who teaches it, and when it is taken. The records are kept after each semester so the enrollment history is not lost. The \texttt{prerequisites} table connects a course to the courses that must be completed before it.

\texttt{CourseGrades} stores the credits and weight received for every course attempt. A student can retake a course in another semester or academic year, and the latest attempt is used when calculating the cumulative GPA. The foreign keys connect these records to the correct student, course, and professor.

\subsection{SQL DDL Commands}

\begin{lstlisting}[breaklines=true]
CREATE TABLE students (
    nim VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    semester INT NOT NULL
        CHECK (semester > 0),
    cohort VARCHAR(4) NOT NULL
        CHECK (cohort ~ '^[0-9]{4}$'),
    program VARCHAR(100) NOT NULL
);

CREATE TABLE professors (
    prof_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE courses (
    course_id VARCHAR(20) PRIMARY KEY,
    course_name VARCHAR(100) NOT NULL,
    period VARCHAR(10) NOT NULL
        CHECK (period IN ('Odd', 'Even', 'Both')),
    type VARCHAR(1) NOT NULL
        CHECK (type IN ('E', 'C'))
);

CREATE TABLE prerequisites (
    prerequisite VARCHAR(20) NOT NULL,
    course_id VARCHAR(20) NOT NULL,

    PRIMARY KEY (prerequisite, course_id),

    FOREIGN KEY (prerequisite)
        REFERENCES courses(course_id),

    FOREIGN KEY (course_id)
        REFERENCES courses(course_id),

    CHECK (prerequisite <> course_id)
);

CREATE TABLE AvailableClasses (
    prof_id VARCHAR(20) NOT NULL,
    nim VARCHAR(20) NOT NULL,
    class VARCHAR(20) NOT NULL,
    course_id VARCHAR(20) NOT NULL,
    semester VARCHAR(10) NOT NULL
        CHECK (semester IN ('Odd', 'Even')),
    academic_year VARCHAR(9) NOT NULL
        CHECK (academic_year ~ '^[0-9]{4}/[0-9]{4}$'),

    PRIMARY KEY (
        nim,
        course_id,
        semester,
        academic_year
    ),

    FOREIGN KEY (prof_id)
        REFERENCES professors(prof_id),

    FOREIGN KEY (nim)
        REFERENCES students(nim),

    FOREIGN KEY (course_id)
        REFERENCES courses(course_id)
);

CREATE TABLE CourseGrades (
    course_id VARCHAR(20) NOT NULL,
    nim VARCHAR(20) NOT NULL,
    academic_year VARCHAR(9) NOT NULL
        CHECK (academic_year ~ '^[0-9]{4}/[0-9]{4}$'),
    completed_period VARCHAR(10) NOT NULL
        CHECK (completed_period IN ('Odd', 'Even')),
    credits SMALLINT NOT NULL
        CHECK (credits > 0),
    weight DECIMAL(3,2) NOT NULL
        CHECK (weight BETWEEN 0.00 AND 4.00),

    PRIMARY KEY (
        course_id,
        nim,
        academic_year,
        completed_period
    ),

    FOREIGN KEY (course_id)
        REFERENCES courses(course_id),

    FOREIGN KEY (nim)
        REFERENCES students(nim),

    FOREIGN KEY (
        nim,
        course_id,
        completed_period,
        academic_year
    )
        REFERENCES AvailableClasses (
            nim,
            course_id,
            semester,
            academic_year
        )
);
\end{lstlisting}

\section{Business Questions and Solutions Using SQL}

\subsection{Question number 1}

\begin{tcolorbox}
  \textit{List of elective course names and the courses that are their prerequisites (only name and course code information is sufficient). Courses that do not have prerequisites should still be displayed!}
\end{tcolorbox}

\begin{enumerate}
    \itemsep0em
    \item Tables needed to answer the query:
    \begin{itemize}
        \item \texttt{courses} (aliased as \texttt{c}): stores the course code, course name, and type.
        \item \texttt{prerequisites} (aliased as \texttt{p}): stores the prerequisite course code for each course.
        \item \texttt{courses} (aliased as \texttt{pre\_c}): used again to get the name of the prerequisite course.
    \end{itemize}

    \item Join criteria (if any):
    \begin{itemize}
        \item \texttt{c.course\_id = p.course\_id}
        \item \texttt{p.prerequisite = pre\_c.course\_id}
    \end{itemize}
    A \texttt{LEFT JOIN} is used so elective courses without prerequisites are still shown.

    \item SQL command according to the question:
    \begin{lstlisting}
SELECT
    c.course_id AS elective_course_code,
    c.course_name AS elective_course_name,
    p.prerequisite AS prerequisite_course_code,
    pre_c.course_name AS prerequisite_course_name
FROM courses c
LEFT JOIN prerequisites p
    ON c.course_id = p.course_id
LEFT JOIN courses pre_c
    ON p.prerequisite = pre_c.course_id
WHERE c.type = 'E'
ORDER BY c.course_id, p.prerequisite;
    \end{lstlisting}

    \item Relational Algebra:

    \begin{align*}
    E &\leftarrow
    \sigma_{c.\text{type}='E'}
    (\rho_c(\text{courses})) \\[4pt]
    J_1 &\leftarrow
    E \leftouterjoin_{
        c.\text{course\_id}=p.\text{course\_id}
    }
    \rho_p(\text{prerequisites}) \\[4pt]
    J_2 &\leftarrow
    J_1 \leftouterjoin_{
        p.\text{prerequisite}=\text{pre\_c.course\_id}
    }
    \rho_{\text{pre\_c}}(\text{courses}) \\[4pt]
    R &\leftarrow
    \tau_{c.\text{course\_id}\uparrow,\,
           p.\text{prerequisite}\uparrow}
    \Big(
    \pi_{\substack{
        c.\text{course\_id},
        c.\text{course\_name},\\
        p.\text{prerequisite},
        \text{pre\_c.course\_name}
    }}(J_2)
    \Big)
    \end{align*}

    \item Alternative SQL commands (if any) with adequate explanation: No alternative query.
\end{enumerate}

\subsection{Question number 2}

\begin{tcolorbox}
List of elective courses taken by Computer Science students of the '2024' cohort in the odd semester of academic year 2026/2027!
\end{tcolorbox}

\begin{enumerate}
    \itemsep0em
    \item Tables needed to answer the query:
    \begin{itemize}
        \item \texttt{students} (aliased as \texttt{s}): stores student information, including their program and cohort.
        \item \texttt{AvailableClasses} (aliased as \texttt{ac}): stores classes taken by students, including the semester and academic year.
        \item \texttt{courses} (aliased as \texttt{c}): stores the course code, course name, and type.
    \end{itemize}

    \item Join criteria (if any):
    \begin{itemize}
        \item \texttt{s.nim = ac.nim}
        \item \texttt{ac.course\_id = c.course\_id}
    \end{itemize}

    \item SQL command according to the question:
    \begin{lstlisting}
SELECT DISTINCT
    c.course_id,
    c.course_name
FROM students s
JOIN AvailableClasses ac
    ON s.nim = ac.nim
JOIN courses c
    ON ac.course_id = c.course_id
WHERE LOWER(s.program) = 'computer science'
    AND s.cohort = '2024'
    AND c.type = 'E'
    AND ac.academic_year = '2026/2027'
    AND ac.semester = 'Odd'
ORDER BY c.course_id;
    \end{lstlisting}

    \item Relational Algebra:

    \begin{align*}
    S &\leftarrow
    \sigma_{\substack{
        \operatorname{LOWER}(s.\text{program})
        =\text{'computer science'}\\
        \land\ s.\text{cohort}=\text{'2024'}
    }}(\rho_s(\text{students})) \\[4pt]
    A &\leftarrow
    \sigma_{\substack{
        \text{ac.academic\_year}=\text{'2026/2027'}\\
        \land\ \text{ac.semester}=\text{'Odd'}
    }}(\rho_{ac}(\text{AvailableClasses})) \\[4pt]
    C &\leftarrow
    \sigma_{c.\text{type}='E'}
    (\rho_c(\text{courses})) \\[4pt]
    J &\leftarrow
    (S \bowtie_{s.\text{nim}=ac.\text{nim}} A)
    \bowtie_{ac.\text{course\_id}=c.\text{course\_id}} C
    \\[4pt]
    R &\leftarrow
    \tau_{c.\text{course\_id}\uparrow}
    \Big(
    \pi_{c.\text{course\_id},c.\text{course\_name}}(J)
    \Big)
    \end{align*}

    \item Alternative SQL commands (if any) with adequate explanation: No alternative query. The semester is checked using \texttt{ac.semester} because it shows when the student took the course.
\end{enumerate}

\subsection{Question number 3}

\begin{tcolorbox}
    List of favorite elective courses for Computer Science students, namely the three courses with the most enrollees in the last three years!
\end{tcolorbox}

\begin{enumerate}
    \itemsep0em
    \item Tables needed to answer the query:
    \begin{itemize}
        \item \texttt{courses} (aliased as \texttt{c}): stores the course code, course name, and type.
        \item \texttt{students} (aliased as \texttt{s}): used to select Computer Science students.
        \item \texttt{AvailableClasses} (aliased as \texttt{ac}): stores student enrollments and their academic years.
    \end{itemize}

    \item Join criteria (if any):
    \begin{itemize}
        \item \texttt{ac.nim = s.nim}
        \item \texttt{ac.course\_id = c.course\_id}
    \end{itemize}

    \item SQL command according to the question:
    \begin{lstlisting}
SELECT
    c.course_id,
    c.course_name,
    COUNT(DISTINCT ac.nim) AS total_enrollees
FROM AvailableClasses ac
JOIN students s
    ON ac.nim = s.nim
JOIN courses c
    ON ac.course_id = c.course_id
WHERE LOWER(s.program) = 'computer science'
    AND c.type = 'E'
    AND CAST(SUBSTRING(ac.academic_year, 1, 4) AS INT) >= (
        SELECT
            MAX(CAST(SUBSTRING(academic_year, 1, 4) AS INT)) - 2
        FROM AvailableClasses
    )
GROUP BY c.course_id, c.course_name
ORDER BY total_enrollees DESC, c.course_id
LIMIT 3;
    \end{lstlisting}

    \item Relational Algebra:

    \begin{align*}
    Y &\leftarrow
    \gamma_{\substack{
        \operatorname{MAX}(
        \operatorname{startYear}(\text{academic\_year})
        )\\
        \rightarrow \text{latestYear}
    }}(\text{AvailableClasses}) \\[4pt]
    J &\leftarrow
    \Big(
    \rho_{ac}(\text{AvailableClasses})
    \bowtie_{ac.\text{nim}=s.\text{nim}}
    \rho_s(\text{students})
    \Big) \\[-2pt]
    &\qquad
    \bowtie_{ac.\text{course\_id}=c.\text{course\_id}}
    \rho_c(\text{courses}) \\[4pt]
    F &\leftarrow
    \sigma_{\substack{
        \operatorname{LOWER}(s.\text{program})
        =\text{'computer science'}\\
        \land\ c.\text{type}='E'\\
        \land\
        \operatorname{startYear}
        (ac.\text{academic\_year})
        \ge \text{latestYear}-2
    }}(J \times Y) \\[4pt]
    G &\leftarrow
    {}_{c.\text{course\_id},c.\text{course\_name}}
    \gamma_{\substack{
        \operatorname{COUNT}(
        \operatorname{DISTINCT}\ ac.\text{nim}
        )\\
        \rightarrow \text{total\_enrollees}
    }}(F) \\[4pt]
    R &\leftarrow
    \operatorname{Top}_3
    \Big(
    \tau_{\text{total\_enrollees}\downarrow,\,
           c.\text{course\_id}\uparrow}(G)
    \Big)
    \end{align*}

    Here, \texttt{startYear} takes the first four digits of the academic year. The latest year is used to select the last three academic years. \(\operatorname{Top}_3\) means only the first three courses are selected after sorting.

    \item Alternative SQL commands (if any) with adequate explanation: No alternative query. The subquery finds the latest academic year in the table, and subtracting two includes a total of three academic years.
\end{enumerate}

\subsection{Question number 4}

\begin{tcolorbox}
List of students who have a Semester Grade Point Average for the odd semester of 2025/2026 of no less than 3.25! (understand the difference between Semester GPA and Cumulative GPA)
\end{tcolorbox}

\begin{enumerate}
    \itemsep0em
    \item Tables needed to answer the query:
    \begin{itemize}
        \item \texttt{students} (aliased as \texttt{s}): stores the student's NIM and name.
        \item \texttt{CourseGrades} (aliased as \texttt{cg}): stores the credits, grade weight, academic year, and completed period.
    \end{itemize}

    \item Join criteria (if any): \texttt{s.nim = cg.nim}

    \item SQL command according to the question:
    \begin{lstlisting}
SELECT
    s.nim,
    s.name,
    ROUND(
        SUM(cg.credits * cg.weight) / SUM(cg.credits),
        2
    ) AS semester_gpa
FROM students s
JOIN CourseGrades cg
    ON s.nim = cg.nim
WHERE cg.academic_year = '2025/2026'
    AND cg.completed_period = 'Odd'
GROUP BY s.nim, s.name
HAVING SUM(cg.credits * cg.weight) / SUM(cg.credits) >= 3.25
ORDER BY semester_gpa DESC, s.nim;
    \end{lstlisting}

    \item Relational Algebra:

    \begin{align*}
    J &\leftarrow
    \rho_s(\text{students})
    \bowtie_{s.\text{nim}=cg.\text{nim}}
    \sigma_{\substack{
        cg.\text{academic\_year}=\text{'2025/2026'}\\
        \land\
        cg.\text{completed\_period}=\text{'Odd'}
    }}(\rho_{cg}(\text{CourseGrades})) \\[4pt]
    G &\leftarrow
    {}_{s.\text{nim},s.\text{name}}
    \gamma_{\substack{
        \operatorname{SUM}(
        cg.\text{credits}\cdot cg.\text{weight}
        )\rightarrow\text{points},\\
        \operatorname{SUM}(cg.\text{credits})
        \rightarrow\text{totalCredits}
    }}(J) \\[4pt]
    F &\leftarrow
    \sigma_{\text{points}/\text{totalCredits}\ge 3.25}
    (G) \\[4pt]
    R &\leftarrow
    \tau_{\text{semester\_gpa}\downarrow,\,
           s.\text{nim}\uparrow}
    \Big(
    \pi_{\substack{
        s.\text{nim},s.\text{name},\\
        \operatorname{ROUND}(
        \text{points}/\text{totalCredits},2)
        \rightarrow\text{semester\_gpa}
    }}(F)
    \Big)
    \end{align*}

    The GPA is calculated using the course credits and grade weights. Students with a GPA of at least 3.25 are selected before the result is rounded to two decimal places.

    \item Alternative SQL commands (if any) with adequate explanation: No alternative query. Only grades from the odd semester of 2025/2026 are used, so the result is a Semester GPA rather than a Cumulative GPA.
\end{enumerate}

\subsection{Question number 5}

\begin{tcolorbox}
    List of students and the Cumulative GPA they have obtained to date! (courses that have been retaken only need to be counted once)
\end{tcolorbox}

\begin{enumerate}
    \itemsep0em
    \item Tables needed to answer the query:
    \begin{itemize}
        \item \texttt{students} (aliased as \texttt{s}): stores the student's NIM and name.
        \item \texttt{CourseGrades} (aliased as \texttt{cg}): stores all course grades obtained by each student, including the academic year and semester.
        \item \texttt{LatestCourseGrades}: selects the latest completed attempt when a student has taken the same course more than once.
    \end{itemize}

    \item Join criteria (if any): \texttt{s.nim = lcg.nim}. Only rows with \texttt{grade\_rank = 1} are used so a repeated course is counted once.

    \item SQL command according to the question:
    \begin{lstlisting}
WITH LatestCourseGrades AS (
    SELECT
        cg.nim,
        cg.course_id,
        cg.credits,
        cg.weight,
        ROW_NUMBER() OVER (
            PARTITION BY cg.nim, cg.course_id
            ORDER BY
                cg.academic_year DESC,
                CASE cg.completed_period
                    WHEN 'Even' THEN 2
                    WHEN 'Odd' THEN 1
                END DESC
        ) AS grade_rank
    FROM CourseGrades cg
    WHERE cg.weight IS NOT NULL
)
SELECT
    s.nim,
    s.name,
    ROUND(
        SUM(lcg.credits * lcg.weight) / SUM(lcg.credits),
        2
    ) AS cumulative_gpa
FROM students s
JOIN LatestCourseGrades lcg
    ON s.nim = lcg.nim
    AND lcg.grade_rank = 1
GROUP BY s.nim, s.name
ORDER BY s.nim;
    \end{lstlisting}

    \item Relational Algebra:

    \begin{align*}
    C &\leftarrow
    \sigma_{cg.\text{weight}\neq\text{NULL}}
    (\rho_{cg}(\text{CourseGrades})) \\[4pt]
    W &\leftarrow
    \omega_{\substack{
        \text{ROW\_NUMBER}()\rightarrow\text{grade\_rank}\\
        \text{PARTITION BY }cg.\text{nim},
        cg.\text{course\_id}\\
        \text{ORDER BY }cg.\text{academic\_year}
        \downarrow,\\
        \operatorname{semesterRank}
        (cg.\text{completed\_period})\downarrow
    }}(C) \\[4pt]
    L &\leftarrow
    \rho_{lcg}(
    \sigma_{\text{grade\_rank}=1}(W)
    ) \\[4pt]
    J &\leftarrow
    \rho_s(\text{students})
    \bowtie_{s.\text{nim}=lcg.\text{nim}}L \\[4pt]
    G &\leftarrow
    {}_{s.\text{nim},s.\text{name}}
    \gamma_{\substack{
        \operatorname{SUM}(
        lcg.\text{credits}\cdot lcg.\text{weight}
        )\rightarrow\text{points},\\
        \operatorname{SUM}(lcg.\text{credits})
        \rightarrow\text{totalCredits}
    }}(J) \\[4pt]
    R &\leftarrow
    \tau_{s.\text{nim}\uparrow}
    \Big(
    \pi_{\substack{
        s.\text{nim},s.\text{name},\\
        \operatorname{ROUND}(
        \text{points}/\text{totalCredits},2)
        \rightarrow\text{cumulative\_gpa}
    }}(G)
    \Big)
    \end{align*}

    Here, \texttt{semesterRank} assigns Odd = 1 and Even = 2. This makes sure the latest semester is selected first when a student retakes a course.

    \item Alternative SQL commands (if any) with adequate explanation: No alternative query. If a course was taken more than once, only the grade from the latest completed attempt is used. The academic year and semester are checked to find the latest attempt, so repeated courses are only counted once when calculating the Cumulative GPA.
\end{enumerate}
